cmake_minimum_required(VERSION 3.21)
project(solohost_browser_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED CEF_ROOT)
  message(FATAL_ERROR "CEF_ROOT is required")
endif()
set(CMAKE_MODULE_PATH ${CEF_ROOT}/cmake ${CMAKE_MODULE_PATH})
find_package(CEF REQUIRED)
find_package(Threads REQUIRED)
find_package(JPEG REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
add_executable(solohost-browser-core
  main.cpp browser_app.cpp browser_app.h browser_client.cpp browser_client.h
  display_server.cpp display_server.h json_util.cpp json_util.h)
add_dependencies(solohost-browser-core libcef_dll_wrapper)
target_include_directories(solohost-browser-core PRIVATE ${CEF_INCLUDE_PATH} ${Boost_INCLUDE_DIRS})
target_link_libraries(solohost-browser-core PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS} JPEG::JPEG Boost::system Threads::Threads dl)
SET_EXECUTABLE_TARGET_PROPERTIES(solohost-browser-core)
SET_CEF_TARGET_OUT_DIR()
